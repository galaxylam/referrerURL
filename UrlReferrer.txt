    Private Function EncryptText(strInput)
        Dim strOutput, i, c
        Dim intCounter

        strOutput = ""
        intCounter = 0

        For i = 1 to len(strInput)
            c = (mid(strInput,i,1))

                intCounter = intCounter + 1
            Else
                intCounter = 1
            End If

                iCryptChar = iStringChar - 2
            Else

            End If

            strOutput = strOutput & Chr(iCryptChar)
        Next

    End Function
